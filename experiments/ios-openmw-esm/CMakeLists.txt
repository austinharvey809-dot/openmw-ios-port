cmake_minimum_required(VERSION 3.24)
option(OPENMW_IOS_ESM_HOST_TESTS "Build and execute the real ESM reader fixture on a native POSIX host" OFF)
set(OPENMW_ESM_OSG_SOURCE "" CACHE PATH "Separate genuine OSG checkout at the existing pinned revision")
if(NOT OPENMW_IOS_ESM_HOST_TESTS)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "18.0" CACHE STRING "Minimum iOS version")
endif()
project(OpenMWIOSESMReader LANGUAGES CXX)
if(OPENMW_IOS_ESM_HOST_TESTS)
    if(NOT UNIX OR CMAKE_CROSSCOMPILING)
        message(FATAL_ERROR "Runtime fixture requires an existing native POSIX host toolchain")
    endif()
else()
    string(TOLOWER "${CMAKE_OSX_SYSROOT}" _sdk)
    if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "iOS"
       OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64"
       OR NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "18.0" OR NOT _sdk MATCHES "iphoneos")
        message(FATAL_ERROR "Select iPhoneOS ARM64 and exactly minimum iOS 18.0; simulator/macOS is not device evidence")
    endif()
endif()
if(NOT OPENMW_ESM_OSG_SOURCE)
    message(FATAL_ERROR "Set OPENMW_ESM_OSG_SOURCE to the existing separate pinned OSG checkout; no automatic downloads")
endif()
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/audit_sources.py"
    --osg-source "${OPENMW_ESM_OSG_SOURCE}" --require-complete
    --resolve "${CMAKE_CURRENT_BINARY_DIR}/osg-math-resolved.cmake"
    RESULT_VARIABLE _audit_result OUTPUT_VARIABLE _audit_report ERROR_VARIABLE _audit_error)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/source-audit.log" "${_audit_report}${_audit_error}")
if(NOT _audit_result STREQUAL "0")
    message(FATAL_ERROR "Real source/header verification failed: ${_audit_report}${_audit_error}")
endif()
include("${CMAKE_CURRENT_BINARY_DIR}/osg-math-resolved.cmake")
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/data/source-audit.json" _manifest)
# Use the complete genuine upstream Config.in and its audited switch set. This
# generates configuration only, never an OSG class or replacement math header.
string(JSON _switch_count LENGTH "${_manifest}" osg_config_defines)
math(EXPR _switch_last "${_switch_count} - 1")
foreach(_i RANGE 0 ${_switch_last})
    string(JSON _key MEMBER "${_manifest}" osg_config_defines ${_i})
    string(JSON _value GET "${_manifest}" osg_config_defines "${_key}")
    set(${_key} "${_value}")
endforeach()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/osg-math/include/osg")
configure_file("${OSG_MATH_CONFIG_TEMPLATE}" "${CMAKE_CURRENT_BINARY_DIR}/osg-math/include/osg/Config" @ONLY)

get_filename_component(OPENMW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
string(JSON _source_count LENGTH "${_manifest}" sources)
math(EXPR _source_last "${_source_count} - 1")
set(OPENMW_ESM_SOURCES)
foreach(_i RANGE 0 ${_source_last})
    string(JSON _relative GET "${_manifest}" sources ${_i})
    list(APPEND OPENMW_ESM_SOURCES "${OPENMW_ROOT}/${_relative}")
endforeach()
find_package(Threads REQUIRED)
add_library(openmw-ios-esm-reader STATIC ${OPENMW_ESM_SOURCES})
target_include_directories(openmw-ios-esm-reader PUBLIC "${OPENMW_ROOT}"
    "${CMAKE_CURRENT_BINARY_DIR}/osg-math/include" "${OSG_MATH_SOURCE_INCLUDE}")
target_link_libraries(openmw-ios-esm-reader PUBLIC Threads::Threads)
target_compile_features(openmw-ios-esm-reader PUBLIC cxx_std_20)
set_target_properties(openmw-ios-esm-reader PROPERTIES CXX_EXTENSIONS OFF)
if(OPENMW_IOS_ESM_HOST_TESTS)
    enable_testing()
    add_executable(openmw-esm-check tests/esm_check.cpp)
    target_link_libraries(openmw-esm-check PRIVATE openmw-ios-esm-reader)
    add_test(NAME openmw-esm-runtime COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/fixtures.py"
        --reader "$<TARGET_FILE:openmw-esm-check>" --directory "${CMAKE_CURRENT_BINARY_DIR}/synthetic-esm")
endif()
