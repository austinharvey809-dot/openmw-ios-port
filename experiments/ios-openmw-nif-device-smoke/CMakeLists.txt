cmake_minimum_required(VERSION 3.24)
set(CMAKE_OSX_DEPLOYMENT_TARGET "18.0" CACHE STRING "Minimum iOS version")
project(OpenMWNifDeviceSmoke LANGUAGES C CXX OBJCXX)
string(TOLOWER "${CMAKE_OSX_SYSROOT}" _sdk)
if(NOT APPLE OR NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" OR NOT _sdk MATCHES "iphoneos"
   OR NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64" OR NOT CMAKE_OSX_DEPLOYMENT_TARGET STREQUAL "18.0")
    message(FATAL_ERROR "G0C requires iPhoneOS ARM64 with minimum iOS 18.0")
endif()
get_filename_component(OPENMW_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(OPENMW_IOS_ESM_VFS_HOST_TESTS OFF)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../ios-openmw-esm-vfs" "${CMAKE_CURRENT_BINARY_DIR}/esm-vfs" EXCLUDE_FROM_ALL)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare.py" audit
    --osg-source "${OPENMW_ESM_OSG_SOURCE}" RESULT_VARIABLE _audit)
if(NOT _audit STREQUAL "0")
    message(FATAL_ERROR "G0C preserved source / fixture audit failed")
endif()
add_subdirectory(cmake/osg)
include(cmake/bullet.cmake)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../ios-openmw-nif-smoke/data/nif-source-sha256.json" _nif_manifest)
string(JSON _count LENGTH "${_nif_manifest}" sources)
math(EXPR _last "${_count} - 1")
set(_nif_sources)
foreach(_i RANGE 0 ${_last})
    string(JSON _source GET "${_nif_manifest}" sources ${_i})
    list(APPEND _nif_sources "${OPENMW_ROOT}/${_source}")
endforeach()
add_library(g0c-nif-parser STATIC ${_nif_sources})
target_include_directories(g0c-nif-parser BEFORE PUBLIC "$<BUILD_INTERFACE:${G0C_OSG_INCLUDE}>")
target_include_directories(g0c-nif-parser PRIVATE "${G0_BULLET_SOURCE}/src"
    "${OPENMW_ROOT}/extern/smhasher/include" "${OPENMW_ROOT}/extern/smhasher/include/smhasher")
target_compile_definitions(g0c-nif-parser PUBLIC BT_USE_DOUBLE_PRECISION)
target_link_libraries(g0c-nif-parser PUBLIC openmw-ios-esm-vfs BulletCollision LinearMath osg OpenThreads)
target_compile_features(g0c-nif-parser PUBLIC cxx_std_20)
set_target_properties(g0c-nif-parser PROPERTIES CXX_EXTENSIONS OFF)

# The unchanged G0B main is also linked as a real iPhoneOS executable (G0C0).
add_executable(openmw-nif-device-link-check "${CMAKE_CURRENT_SOURCE_DIR}/../ios-openmw-nif-smoke/tests/nif_parse_check.cpp")
target_link_libraries(openmw-nif-device-link-check PRIVATE g0c-nif-parser)
# iPhoneOS may wrap even this link-check target in an .app. Export the actual binary.
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/g0c0-executable-$<CONFIG>.txt"
    CONTENT "$<TARGET_FILE:openmw-nif-device-link-check>\n")
set(_prepared "${CMAKE_CURRENT_BINARY_DIR}/prepared")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tools/prepare.py" prepare
    --osg-source "${OPENMW_ESM_OSG_SOURCE}" --generated "${G0C_OSG_INCLUDE}"
    --bullet-source "${G0_BULLET_SOURCE}" --output "${_prepared}" RESULT_VARIABLE _prepare)
if(NOT _prepare STREQUAL "0")
    message(FATAL_ERROR "G0C resource/provenance preparation failed")
endif()
add_executable(IOSOpenMWNifSmoke MACOSX_BUNDLE src/main.mm src/NifCheck.mm src/NifCheck.hpp src/NifProbe.cpp src/NifProbe.hpp)
add_dependencies(IOSOpenMWNifSmoke openmw-nif-device-link-check)
target_link_libraries(IOSOpenMWNifSmoke PRIVATE g0c-nif-parser)
target_include_directories(IOSOpenMWNifSmoke PRIVATE "${_prepared}/include")
target_compile_options(IOSOpenMWNifSmoke PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>" -Wall -Wextra)
foreach(_framework UIKit Foundation)
    find_library(G0C_${_framework} NAMES ${_framework} REQUIRED)
    target_link_libraries(IOSOpenMWNifSmoke PRIVATE "${G0C_${_framework}}")
endforeach()
foreach(_target IOSOpenMWNifSmoke openmw-nif-device-link-check)
    target_compile_features(${_target} PRIVATE cxx_std_20)
    target_link_options(${_target} PRIVATE "-Wl,-no_adhoc_codesign")
    set_target_properties(${_target} PROPERTIES CXX_EXTENSIONS OFF OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES
        BUILD_RPATH "" INSTALL_RPATH "" SKIP_BUILD_RPATH TRUE INSTALL_RPATH_USE_LINK_PATH FALSE
        XCODE_ATTRIBUTE_LD_RUNPATH_SEARCH_PATHS "" XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO")
endforeach()
set_target_properties(IOSOpenMWNifSmoke PROPERTIES
    MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/Info.plist.in"
    MACOSX_BUNDLE_GUI_IDENTIFIER "org.openmw.experiments.ios-openmw-nif-smoke"
    XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "org.openmw.experiments.ios-openmw-nif-smoke"
    XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1" XCODE_ATTRIBUTE_DEBUG_INFORMATION_FORMAT "dwarf-with-dsym")
file(GLOB_RECURSE _resources CONFIGURE_DEPENDS "${_prepared}/resources/*")
foreach(_path IN LISTS _resources)
    file(RELATIVE_PATH _relative "${_prepared}/resources" "${_path}")
    get_filename_component(_directory "${_relative}" DIRECTORY)
    if(_directory STREQUAL "")
        set(_directory ".")
    endif()
    set_source_files_properties("${_path}" PROPERTIES HEADER_FILE_ONLY TRUE MACOSX_PACKAGE_LOCATION "${_directory}")
    target_sources(IOSOpenMWNifSmoke PRIVATE "${_path}")
endforeach()
