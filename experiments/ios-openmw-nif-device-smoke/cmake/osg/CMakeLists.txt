# Build original core recipes only, avoiding the pinned root's hardcoded Xcode/iOS SDK path.
project(G0COSG LANGUAGES C CXX)
set(PROJECT_SOURCE_DIR "${OPENMW_ESM_OSG_SOURCE}")
set(OpenSceneGraph_SOURCE_DIR "${OPENMW_ESM_OSG_SOURCE}")
set(OpenThreads_SOURCE_DIR "${OPENMW_ESM_OSG_SOURCE}")
list(PREPEND CMAKE_MODULE_PATH "${OPENMW_ESM_OSG_SOURCE}/CMakeModules")
include(GNUInstallDirs)
include(OsgMacroUtils)
find_package(Threads REQUIRED)
find_library(OPENGL_gl_LIBRARY NAMES OpenGLES REQUIRED)
set(DYNAMIC_OPENSCENEGRAPH OFF CACHE BOOL "Static OSG core" FORCE)
set(DYNAMIC_OPENTHREADS OFF CACHE BOOL "Static OpenThreads" FORCE)
set(OPENSCENEGRAPH_USER_DEFINED_DYNAMIC_OR_STATIC STATIC)
set(OSG_COMPILE_FRAMEWORKS OFF)
# Supported upstream fallback avoids try_run and deprecated OSAtomic on iPhoneOS.
set(OPENTHREADS_ATOMIC_USE_MUTEX ON CACHE BOOL "Use genuine pthread mutex atomics" FORCE)
file(READ "${OPENMW_ESM_OSG_SOURCE}/CMakeLists.txt" _upstream)
foreach(_name MAJOR_VERSION MINOR_VERSION PATCH_VERSION SOVERSION RELEASE_CANDIDATE)
    string(REGEX MATCH "SET\\(OPENSCENEGRAPH_${_name} ([0-9]+)\\)" _match "${_upstream}")
    if(NOT _match)
        message(FATAL_ERROR "Pinned OSG version declaration missing")
    endif()
    set(OPENSCENEGRAPH_${_name} "${CMAKE_MATCH_1}")
endforeach()
set(OPENSCENEGRAPH_VERSION "${OPENSCENEGRAPH_MAJOR_VERSION}.${OPENSCENEGRAPH_MINOR_VERSION}.${OPENSCENEGRAPH_PATCH_VERSION}")
file(READ "${OPENMW_ROOT}/experiments/ios-openmw-esm/data/source-audit.json" _d_manifest)
string(JSON _switch_count LENGTH "${_d_manifest}" osg_config_defines)
math(EXPR _switch_last "${_switch_count} - 1")
foreach(_i RANGE 0 ${_switch_last})
    string(JSON _key MEMBER "${_d_manifest}" osg_config_defines ${_i})
    string(JSON _value GET "${_d_manifest}" osg_config_defines "${_key}")
    set(${_key} "${_value}" CACHE BOOL "Audited D math configuration" FORCE)
endforeach()
set(_include "${CMAKE_CURRENT_BINARY_DIR}/include")
set(OPENSCENEGRAPH_CONFIG_HEADER "${_include}/osg/Config")
configure_file("${OPENMW_ESM_OSG_SOURCE}/src/osg/Config.in" "${OPENSCENEGRAPH_CONFIG_HEADER}" @ONLY)
# GLES2 is upstream's Apple-device profile. It supplies SDK headers, no ANGLE.
foreach(_api GL1 GL2 GL3 GLES1 GLES2 GLES3)
    set(OSG_${_api}_FEATURES 0)
    set(OSG_${_api}_AVAILABLE OFF)
endforeach()
set(OSG_GLES2_AVAILABLE ON)
set(OSG_GLES2_FEATURES 1)
set(OSG_GL_CONTEXT_VERSION "2.0")
set(OPENGL_HEADER1 "#include <TargetConditionals.h>")
set(OPENGL_HEADER2 "#include <OpenGLES/ES2/gl.h>")
include(CheckCXXSourceCompiles)
check_cxx_source_compiles("${OPENGL_HEADER2}\nint main(){ GLint64 value=0; return int(value); }" GL_HEADER_HAS_GLINT64)
check_cxx_source_compiles("${OPENGL_HEADER2}\nint main(){ GLuint64 value=0; return int(value); }" GL_HEADER_HAS_GLUINT64)
set(OPENSCENEGRAPH_OPENGL_HEADER "${_include}/osg/GL")
configure_file("${OPENMW_ESM_OSG_SOURCE}/src/osg/GL.in" "${OPENSCENEGRAPH_OPENGL_HEADER}" @ONLY)
set(OPENSCENEGRAPH_VERSION_HEADER "${_include}/osg/Version")
configure_file("${OPENMW_ESM_OSG_SOURCE}/src/osg/Version.in" "${OPENSCENEGRAPH_VERSION_HEADER}" @ONLY)
include_directories("${_include}" "${OPENMW_ESM_OSG_SOURCE}/include")
add_subdirectory("${OPENMW_ESM_OSG_SOURCE}/src/OpenThreads" "${CMAKE_CURRENT_BINARY_DIR}/OpenThreads" EXCLUDE_FROM_ALL)
add_subdirectory("${OPENMW_ESM_OSG_SOURCE}/src/osg" "${CMAKE_CURRENT_BINARY_DIR}/osg" EXCLUDE_FROM_ALL)
foreach(_header OpenThreads/Config OpenThreads/Version osg/Config osg/GL osg/Version)
    if(NOT EXISTS "${_include}/${_header}")
        message(FATAL_ERROR "Missing genuine OSG generated header: ${_header}")
    endif()
endforeach()
foreach(_target osg OpenThreads)
    target_include_directories(${_target} PUBLIC "$<BUILD_INTERFACE:${_include}>" "${OPENMW_ESM_OSG_SOURCE}/include")
endforeach()
set(G0C_OSG_INCLUDE "${_include}" PARENT_SCOPE)
