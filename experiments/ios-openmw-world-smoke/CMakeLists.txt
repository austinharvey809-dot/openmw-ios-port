cmake_minimum_required(VERSION 3.24)
project(OpenMWWorldIntegration LANGUAGES C CXX)
get_filename_component(G4_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" ABSOLUTE)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
if(G4_CPU_PROVIDER)
  if(G4_CPU_AR)
    set(CMAKE_AR "${G4_CPU_AR}")
    set(CMAKE_RANLIB "${G4_CPU_RANLIB}")
  endif()
  include(cmake/windows_provider.cmake)
else()
  # Retain accepted native suites and share exactly one OSG/Bullet provider.
  add_subdirectory("${G4_ROOT}/../ios-openmw-exterior-smoke" "${CMAKE_CURRENT_BINARY_DIR}/provider" EXCLUDE_FROM_ALL)
  set(G4_OSG_SOURCE "${OPENMW_ESM_OSG_SOURCE}")
  get_target_property(_osg_binary osg BINARY_DIR)
  get_filename_component(G4_OSG_GENERATED_INCLUDE "${_osg_binary}/../../include" ABSOLUTE)
endif()
include(cmake/engine.cmake)
add_executable(openmw-world-check tests/world_check.cpp)
target_link_libraries(openmw-world-check PRIVATE g4-upstream)
add_library(g4-device-headers OBJECT tests/device_scene_headers.cpp)
target_link_libraries(g4-device-headers PRIVATE g4-upstream)
add_executable(openmw-g4-osg-registration-check tests/osg_registration.cpp)
target_link_libraries(openmw-g4-osg-registration-check PRIVATE ${G4_REGISTRATION_LIBRARIES})
target_compile_features(openmw-g4-osg-registration-check PRIVATE cxx_std_20)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/g4-products-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:openmw-world-check>\n")
enable_testing()
add_test(NAME g4-osg-static-registration COMMAND openmw-g4-osg-registration-check)
add_test(NAME g4-genuine-world-regressions COMMAND "${Python3_EXECUTABLE}" "${G4_ROOT}/tools/run_cases.py"
  --binary "$<TARGET_FILE:openmw-world-check>" --shaders "${G4_GENERATED}/Shaders")
