cmake_minimum_required(VERSION 3.24)
project(OpenMWWorldIntegration LANGUAGES C CXX)
get_filename_component(G4_ROOT "${CMAKE_CURRENT_SOURCE_DIR}" ABSOLUTE)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}")
if(G4_CPU_PROVIDER)
  if(G4_CPU_AR)
    set(CMAKE_AR "${G4_CPU_AR}")
    set(CMAKE_RANLIB "${G4_CPU_RANLIB}")
  endif()
  include(cmake/windows_provider.cmake)
else()
  # Retain accepted native suites and share exactly one OSG/Bullet provider.
  add_subdirectory("${G4_ROOT}/../ios-openmw-exterior-smoke" "${CMAKE_CURRENT_BINARY_DIR}/provider" EXCLUDE_FROM_ALL)
  set(G4_OSG_SOURCE "${OPENMW_ESM_OSG_SOURCE}")
  get_target_property(_osg_binary osg BINARY_DIR)
  get_filename_component(G4_OSG_GENERATED_INCLUDE "${_osg_binary}/../../include" ABSOLUTE)
endif()
include(cmake/engine.cmake)
# Small host runtime gate: same genuine OSG provider, no world/Bullet/image build.
add_executable(openmw-g4-startup-check tests/startup_check.cpp "${G4_REPO}/components/sceneutil/glextensions.cpp")
target_include_directories(openmw-g4-startup-check BEFORE PRIVATE "${G4_OSG_GENERATED_INCLUDE}")
target_include_directories(openmw-g4-startup-check PRIVATE "${G4_ROOT}/src" "${G4_REPO}" "${G4_OSG_SOURCE}/include")
target_link_libraries(openmw-g4-startup-check PRIVATE osgViewer osgGA osgDB osgUtil osgText osg OpenThreads)
target_compile_features(openmw-g4-startup-check PRIVATE cxx_std_20)
set_target_properties(openmw-g4-startup-check PROPERTIES CXX_EXTENSIONS OFF)
if(NOT G4_SHARED_DESKTOP_PROVIDER)
  target_compile_definitions(openmw-g4-startup-check PRIVATE OSG_LIBRARY_STATIC OPENTHREADS_STATIC)
endif()
if(WIN32)
  target_link_libraries(openmw-g4-startup-check PRIVATE opengl32)
elseif(APPLE)
  find_library(G4_HOST_OPENGL OpenGL REQUIRED)
  target_link_libraries(openmw-g4-startup-check PRIVATE "${G4_HOST_OPENGL}")
else()
  find_package(OpenGL REQUIRED)
  target_link_libraries(openmw-g4-startup-check PRIVATE OpenGL::GL)
endif()
add_executable(openmw-world-check tests/world_check.cpp)
target_link_libraries(openmw-world-check PRIVATE g4-upstream)
add_library(g4-device-headers OBJECT tests/device_scene_headers.cpp)
target_link_libraries(g4-device-headers PRIVATE g4-upstream)
add_executable(openmw-g4-osg-registration-check tests/osg_registration.cpp)
target_link_libraries(openmw-g4-osg-registration-check PRIVATE ${G4_REGISTRATION_LIBRARIES})
target_compile_features(openmw-g4-osg-registration-check PRIVATE cxx_std_20)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/g4-products-$<CONFIG>.txt" CONTENT "$<TARGET_FILE:openmw-world-check>\n")
enable_testing()
add_test(NAME g4-startup-capabilities COMMAND "${Python3_EXECUTABLE}" "${G4_ROOT}/tools/run_cases.py"
  --startup-binary "$<TARGET_FILE:openmw-g4-startup-check>" --output "${CMAKE_CURRENT_BINARY_DIR}/g4-startup-case.json")
add_test(NAME g4-osg-static-registration COMMAND openmw-g4-osg-registration-check)
add_test(NAME g4-genuine-world-regressions COMMAND "${Python3_EXECUTABLE}" "${G4_ROOT}/tools/run_cases.py"
  --binary "$<TARGET_FILE:openmw-world-check>" --shaders "${G4_GENERATED}/Shaders"
  --output "${CMAKE_CURRENT_BINARY_DIR}/g4-world-cases.json")
set_tests_properties(g4-genuine-world-regressions PROPERTIES DEPENDS g4-startup-capabilities)
